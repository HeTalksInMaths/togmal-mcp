ACTUAL EXTRACTION CODE (Simplified):

# From extract_model_performance.py:

for eval_file in glob("data/eval_cache/mmlu_pro_*.json"):
    data = json.load(eval_file)
    
    model_name = clean_model_name(data["model_name"])
    
    if not data.get("predictions"):
        continue  # Skip files with no predictions
    
    for pred in data["predictions"]:
        qid = pred["question_id"]
        
        all_questions[qid][model_name] = {
            "question_text": pred["question"],
            "correct_answer": pred["answer"],
            "model_prediction": pred["pred"],
            "is_correct": pred["pred"] == pred["answer"]
        }

# Result:
# - 4 files had predictions
# - 170 unique question IDs
# - 4 models tracked
